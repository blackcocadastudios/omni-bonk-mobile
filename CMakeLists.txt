cmake_minimum_required(VERSION 3.10)
project(omnibonk CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")

file(GLOB GAME_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
list(FILTER GAME_SRC EXCLUDE REGEX ".*main_(android|desktop|sdl)\\.cpp$")
option(OMNIBONK_SDL "Build the SDL2 port (Linux handhelds such as the R36S, or desktop)" OFF)

if(ANDROID)
  # Native activity shared library (built by scripts/build_apk.sh with the NDK toolchain)
  set(GLUE ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
  add_library(omnibonk SHARED ${GAME_SRC} src/main_android.cpp ${GLUE})
  target_include_directories(omnibonk PRIVATE ${ANDROID_NDK}/sources/android/native_app_glue)
  target_compile_options(omnibonk PRIVATE -ffunction-sections -fdata-sections -Wno-unused-parameter)
  set_target_properties(omnibonk PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate -Wl,--gc-sections -Wl,--build-id=sha1 -Wl,-z,max-page-size=16384")
  target_link_libraries(omnibonk android log EGL GLESv3 aaudio m)
elseif(OMNIBONK_SDL)
  # SDL2 + OpenGL ES 3.0 executable (scripts/build_r36s.sh cross-compiles it for aarch64).
  # GL entry points are loaded at runtime through SDL, so the only non-libc dependency is libSDL2-2.0.so.0.
  if(NOT SDL2_INCLUDE_DIR OR NOT SDL2_LIBRARY)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2PC REQUIRED sdl2)
    set(SDL2_INCLUDE_DIR ${SDL2PC_INCLUDE_DIRS})
    set(SDL2_LIBRARY ${SDL2PC_LINK_LIBRARIES})
  endif()
  add_executable(omnibonk ${GAME_SRC} src/main_sdl.cpp)
  target_compile_definitions(omnibonk PRIVATE OMNIBONK_GL_LOADER _REENTRANT)
  target_include_directories(omnibonk PRIVATE ${SDL2_INCLUDE_DIR})
  target_compile_options(omnibonk PRIVATE -fno-math-errno -ffunction-sections -fdata-sections -Wall -Wno-unused-parameter)
  target_link_libraries(omnibonk ${SDL2_LIBRARY} m dl pthread -static-libstdc++ -static-libgcc -Wl,--gc-sections -Wl,--as-needed)
else()
  add_executable(omnibonk_desktop ${GAME_SRC} src/main_desktop.cpp)
  target_link_libraries(omnibonk_desktop /usr/lib/x86_64-linux-gnu/libGLESv2.so.2 /usr/lib/x86_64-linux-gnu/libEGL.so.1 png z pthread)
endif()
